import java.util.ArrayList;
import java.util.List;

public class Solution {
public List<String> findAndReplacePattern(String[] words, String pattern) {
List<String> result = new ArrayList<>();

for (String word : words) {
if (matches(word, pattern)) {
result.add(word);
}
}

return result;
}

private boolean matches(String word, String pattern) {
// Arrays to act as maps for characters (size 26 for lowercase English letters)
int[] wordToPattern = new int[26];
int[] patternToWord = new int[26];

for (int i = 0; i < word.length(); i++) {
int wChar = word.charAt(i) - 'a';
int pChar = pattern.charAt(i) - 'a';

// If neither character has been mapped yet, create the two-way mapping
// (We add 1 so that 'a' mapped to 'a' isn't confused with the default 0)
if (wordToPattern[wChar] == 0 && patternToWord[pChar] == 0) {
wordToPattern[wChar] = pChar + 1;
patternToWord[pChar] = wChar + 1;
}
// If they are mapped, they must match each other exactly
else if (wordToPattern[wChar] != pChar + 1 || patternToWord[pChar] != wChar + 1) {
return false;
}
}

return true;
}
}
