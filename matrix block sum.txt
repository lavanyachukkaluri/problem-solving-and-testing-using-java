class Solution {
public int[][] matrixBlockSum(int[][] mat, int k) {
int m = mat.length;
int n = mat[0].length;

// Create a 2D prefix sum array with extra padding row and column for easy boundary handling
int[][] s = new int[m + 1][n + 1];

// Step 1: Build the prefix sum matrix
for (int i = 0; i < m; ++i) {
for (int j = 0; j < n; ++j) {
s[i + 1][j + 1] = s[i][j + 1] + s[i + 1][j] - s[i][j] + mat[i][j];
}
}

int[][] ans = new int[m][n];

// Step 2: Calculate the block sum for each element using the prefix sum matrix
for (int i = 0; i < m; ++i) {
for (int j = 0; j < n; ++j) {
// Determine the valid boundaries of the block
int x1 = Math.max(i - k, 0);
int y1 = Math.max(j - k, 0);
int x2 = Math.min(m - 1, i + k);
int y2 = Math.min(n - 1, j + k);

// Query the prefix sum matrix in O(1) time using inclusion-exclusion principle
ans[i][j] = s[x2 + 1][y2 + 1] - s[x1][y2 + 1] - s[x2 + 1][y1] + s[x1][y1];
}
}

return ans;
}
}
