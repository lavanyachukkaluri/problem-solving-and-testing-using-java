class Solution {
public boolean halvesAreAlike(String s) {
String vowels = "aeiouAEIOU";
int n = s.length() / 2;

int first = 0, second = 0;

for (int i = 0; i < n; i++) {
if (vowels.indexOf(s.charAt(i)) != -1) {
first++;
}
}

for (int i = n; i < s.length(); i++) {
if (vowels.indexOf(s.charAt(i)) != -1) {
second++;
}
}

return first == second;
}
}
